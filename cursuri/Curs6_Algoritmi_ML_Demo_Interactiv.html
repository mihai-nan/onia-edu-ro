Exemplu Python cu scikit-learn 
import pandas as pd
from sklearn.compose import ColumnTransformer
from sklearn.preprocessing import OneHotEncoder
from sklearn.tree import DecisionTreeClassifier
from sklearn.pipeline import make_pipeline

# Exemplul cu 5 clienți din curs
X = pd.DataFrame({
    "venit": ["mare", "mediu", "mare", "mic", "mediu"],
    "datorii": ["mici", "mari", "mari", "mari", "mici"],
    "vechime": [5, 2, 8, 1, 4],
    "intarzieri": ["nu", "da", "nu", "da", "nu"]
})
y = ["Aprobat", "Respins", "Aprobat", "Respins", "Aprobat"]

categorice = ["venit", "datorii", "intarzieri"]
preprocesare = ColumnTransformer([
    ("cat", OneHotEncoder(handle_unknown="ignore"), categorice)
], remainder="passthrough")

model = make_pipeline(preprocesare, DecisionTreeClassifier(
    criterion="gini", max_depth=3, random_state=42
))
model.fit(X, y)
client_nou = pd.DataFrame([{
    "venit": "mediu", "datorii": "mici",
    "vechime": 3, "intarzieri": "nu"
}])
print(model.predict(client_nou))

# Atenție: figura conceptuală poate avea ramuri multiple
# pentru categorii; DecisionTreeClassifier realizează
# teste binare pe caracteristicile codificate numeric.

from sklearn.neighbors import KNeighborsClassifier

# Cele 8 puncte din exemplul cursului

Exemplu Python cu scikit-learn 
import numpy as np
from sklearn.linear_model import LinearRegression
from sklearn.metrics import mean_absolute_error, mean_squared_error

# Apartamentele din curs: suprafață, preț (mii lei)
X = np.array([[40], [50], [60], [70], [80], [90]])
y = np.array([240, 285, 330, 390, 430, 485])

model = LinearRegression()
model.fit(X, y)

print("a =", model.coef_[0])
print("b =", model.intercept_)
print("Estimare pentru 75 m² =", model.predict([[75]])[0])

predictii = model.predict(X)
print("MAE =", mean_absolute_error(y, predictii))
print("MSE =", mean_squared_error(y, predictii))

# Pentru subsetul cu 4 observații (40, 50, 60, 70),
# dreapta exactă din curs este y_hat = 4.95 * x + 39.

Exemplele de antrenare

Evoluția pierderii MSE după fiecare epocă

Cod Python: actualizare prin gradient descendent stocastic 
import numpy as np

# O actualizare stocastică (un exemplu la un moment dat)
X = np.array([1., 2., 3., 4., 5., 6.])
y = np.array([2.1, 3.5, 4.8, 6.1, 7.9, 9.0])

# Standardizăm pentru stabilitatea gradientului.
mx, sx = X.mean(), X.std() or 1
my, sy = y.mean(), y.std() or 1
u, v = (X-mx)/sx, (y-my)/sy

rng = np.random.default_rng(17)
a, b = rng.uniform(-1, 1, size=2)
eta = 0.04
for epoca in range(24):
    for xi, yi in zip(u, v):
        eroare = (a*xi + b) - yi
        da, db = 2*eroare*xi, 2*eroare
        a -= eta*da
        b -= eta*db

panta = a*sy/sx
intercept = my + b*sy - panta*mx
print(panta, intercept)

Când o dreaptă nu este suficientă: regresia polinomială

O dreaptă are pantă constantă și nu poate urmări o curbă în formă de U. Cu PolynomialFeatures construim coloane noi (x², x³ etc.), apoi aplicăm aceeași regresie liniară asupra coloanelor rezultate. Urmărește fiecare transformare numerică și efectul ei asupra graficului.

Comparație: dreaptă și curbă

Albastru: date; gri punctat: regresie liniară; turcoaz: modelul cu caracteristici polinomiale.

Din caracteristica originală la caracteristici polinomiale

Fiecare linie este o observație, iar coloanele noi sunt funcții ale lui x.
Cod Python: PolynomialFeatures și LinearRegression 
import numpy as np
from sklearn.pipeline import make_pipeline
from sklearn.preprocessing import PolynomialFeatures
from sklearn.linear_model import LinearRegression
from sklearn.metrics import mean_squared_error

X = np.array([-3, -2, -1, 0, 1, 2, 3]).reshape(-1, 1)
y = np.array([9.3, 4.4, 1.6, 0.3, 1.3, 4.2, 9.6])

linear = LinearRegression().fit(X, y)
polinomial = make_pipeline(
    PolynomialFeatures(degree=2, include_bias=False),
    LinearRegression()
).fit(X, y)

print("MSE dreptă:", mean_squared_error(y, linear.predict(X)))
print("MSE polinomial:", mean_squared_error(y, polinomial.predict(X)))
print("Predicție x=2:", polinomial.predict([[2]])[0])

Bucla de optimizare

Atribuire → medie → repetare până la stabilizare.
Exemplu Python cu scikit-learn 
import numpy as np
from sklearn.cluster import KMeans

# Trei grupuri de puncte, ca în prezentarea cursului
X = np.array([
    [1.0, 1.0], [1.4, 1.5], [1.8, 1.2], [2.0, 1.8], [1.3, 2.0],
    [4.2, 1.0], [4.8, 1.4], [5.1, 1.1], [5.4, 1.7], [4.6, 2.0],
    [6.2, 3.0], [6.8, 3.4], [7.1, 3.1], [7.4, 3.8], [6.5, 4.0]
])

model = KMeans(n_clusters=3, n_init=10, random_state=42)
labels = model.fit_predict(X)
print(labels)                 # grupul atribuit fiecărui punct
print(model.cluster_centers_) # coordonatele centroizilor
print(model.inertia_)         # suma distanțelor pătratice

# În simulator demonstrăm pașii de atribuire și
# actualizare explicit, pentru o inițializare aleasă.
# scikit-learn poate folosi o altă inițializare și alt rezultat.
