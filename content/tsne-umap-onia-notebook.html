Pasul 1 din t-SNE: distanțe și probabilități condiționate

from math import exp
puncte = [[0., 0.], [1., 0.], [3., 0.]]
centru = puncte[1]
sigma = 1.0
ponderi = []
for i, p in enumerate(puncte):
    if p == centru:
        ponderi.append(0.0)
    else:
        d2 = (p[0]-centru[0])**2 + (p[1]-centru[1])**2
        ponderi.append(exp(-d2/(2*sigma**2)))
s = sum(ponderi)
print([round(w/s, 3) for w in ponderi])

Pasul 3: funcția de pierdere și actualizarea coordonatelor

from math import log
P = [0.50, 0.30, 0.20]
Q = [0.45, 0.35, 0.20]
kl = sum(p*log(p/q) for p, q in zip(P, Q))
print(round(kl, 5))

from sklearn.datasets import load_iris
from sklearn.decomposition import PCA
from sklearn.manifold import trustworthiness
from sklearn.preprocessing import StandardScaler
X, _ = load_iris(return_X_y=True)
X = StandardScaler().fit_transform(X)
Z = PCA(n_components=2).fit_transform(X)
print(round(trustworthiness(X, Z, n_neighbors=5), 3))

Implementare didactică t-SNE: un pas de gradient de la zero

import numpy as np

X = np.array([[0., 0.], [0.2, 0.1], [3., 3.], [3.2, 2.9]])
n = len(X)
d2 = ((X[:, None, :] - X[None, :, :]) ** 2).sum(axis=2)
P = np.exp(-d2 / 2.0)
np.fill_diagonal(P, 0)
P = P / P.sum()

rng = np.random.default_rng(0)
Y = rng.normal(0, 0.01, (n, 2))
for pas in range(51):
    delta = Y[:, None, :] - Y[None, :, :]
    dist2_y = (delta ** 2).sum(axis=2)
    inv = 1.0 / (1.0 + dist2_y)
    np.fill_diagonal(inv, 0)
    Q = inv / inv.sum()
    cost = np.sum(P[P > 0] * np.log(P[P > 0] / Q[P > 0]))
    if pas % 10 == 0:
        print(pas, round(cost, 5))
    grad = 4 * np.sum(((P-Q)*inv)[:, :, None] * delta, axis=1)
    Y -= 2.0 * grad
print('Coordonate finale:', Y.round(2))
