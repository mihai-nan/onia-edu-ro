MAE, MSE și RMSE din liste Python Cod Python ▾
from math import sqrt

def metrici(y_real, y_prezis):
    n = len(y_real)
    erori = [y_prezis[i] - y_real[i] for i in range(n)]
    mae = sum(abs(e) for e in erori) / n
    mse = sum(e * e for e in erori) / n
    return mae, mse, sqrt(mse)

print(metrici([3, 4, 5], [4, 2, 5]))
# (1.0, 1.666..., 1.290...)
Soluția exactă OLS pentru o caracteristică (Python simplu) Cod Python ▾
def regresie_ols(x, y):
    n = len(x)
    if n != len(y) or n < 2:
        raise ValueError("Sunt necesare cel puțin două perechi")

    media_x = sum(x) / n
    media_y = sum(y) / n
    sus = sum((x[i] - media_x) * (y[i] - media_y)
              for i in range(n))
    jos = sum((x[i] - media_x) ** 2 for i in range(n))
    if jos == 0:
        raise ValueError("Toate valorile lui x sunt identice")

    w = sus / jos
    b = media_y - w * media_x
    return w, b

print(regresie_ols([1, 2, 3], [2, 3, 5]))
# (1.5, 0.3333333333333335)

Antrenarea din zero, cu Python de bază Cod Python ▾
def antreneaza_linie(x, y, eta=0.05, epoci=2000):
    n = len(x)
    w, b = 0.0, 0.0
    istoric = []

    for epoca in range(epoci):
        gradient_w = 0.0
        gradient_b = 0.0
        eroare_totala = 0.0

        for i in range(n):
            predictie = w * x[i] + b
            eroare = predictie - y[i]
            gradient_w += eroare * x[i]
            gradient_b += eroare
            eroare_totala += eroare ** 2

        gradient_w = 2 * gradient_w / n
        gradient_b = 2 * gradient_b / n
        w -= eta * gradient_w
        b -= eta * gradient_b
        if epoca % 200 == 0:
            istoric.append(eroare_totala / n)

    return w, b, istoric

x = [1, 2, 3]
y = [2, 3, 5]
w, b, erori = antreneaza_linie(x, y, eta=0.05)
print(round(w, 3), round(b, 3))
print([round(e, 3) for e in erori[:4]])

def antreneaza(X, y, eta=0.05, epoci=3000):
    n = len(X)
    k = len(X[0])
    w = [0.0] * k
    b = 0.0

    for epoca in range(epoci):
        gw = [0.0] * k
        gb = 0.0
        for i in range(n):
            predictie = b
            for j in range(k):
                predictie += w[j] * X[i][j]
            eroare = predictie - y[i]
            gb += eroare
            for j in range(k):
                gw[j] += eroare * X[i][j]

        b -= eta * (2 * gb / n)
        for j in range(k):
            w[j] -= eta * (2 * gw[j] / n)
    return w, b

X = [[0, 0], [1, 0], [0, 1], [1, 1]]
y = [2, 5, 4, 7]  # y = 3*x1 + 2*x2 + 2
w, b = antreneaza(X, y)
print([round(v, 2) for v in w], round(b, 2))
# [3.0, 2.0] 2.0

Ridge pentru o caracteristică — implementare de la zero Cod Python ▾
def ridge_1d(x, y, lam):
    n = len(x)
    mx = sum(x) / n
    my = sum(y) / n
    sxy = sum((x[i] - mx) * (y[i] - my)
              for i in range(n))
    sxx = sum((xi - mx) ** 2 for xi in x)
    if sxx + lam == 0:
        raise ValueError("Nu putem determina o pantă unică")
    w = sxy / (sxx + lam)
    b = my - w * mx
    return w, b

x = [1, 2, 3]
y = [2, 3, 5]
print(ridge_1d(x, y, 0))   # aceeași pantă ca OLS
print(ridge_1d(x, y, 2))   # panta este micșorată
Lasso unidimensional — soluție exactă, fără biblioteci Cod Python ▾
def prag_moale(z, prag):
    if z > prag:
        return z - prag
    if z < -prag:
        return z + prag
    return 0.0

def lasso_1d(x, y, lam):
    n = len(x)
    mx, my = sum(x) / n, sum(y) / n
    sxy = sum((x[i] - mx) * (y[i] - my)
              for i in range(n))
    sxx = sum((xi - mx) ** 2 for xi in x)
    if sxx == 0:
        raise ValueError("Toate valorile lui x sunt identice")
    w = prag_moale(sxy, lam / 2) / sxx
    return w, my - w * mx

print(lasso_1d([1, 2, 3], [2, 3, 5], 10))
# (0.0, 3.3333333333333335)
Elastic Net cu o singură caracteristică, de la zero Cod Python ▾
def prag_moale(z, prag):
    if z > prag:
        return z - prag
    if z < -prag:
        return z + prag
    return 0.0

def elastic_net_1d(x, y, lam1, lam2):
    n = len(x)
    mx = sum(x) / n
    my = sum(y) / n
    sxy = sum((x[i] - mx) * (y[i] - my)
              for i in range(n))
    sxx = sum((xi - mx) ** 2 for xi in x)
    w = prag_moale(sxy, lam1 / 2) / (sxx + lam2)
    b = my - w * mx
    return w, b

print(elastic_net_1d([1, 2, 3], [2, 3, 5], 1, 2))
# (0.625, 2.0833333333333335)

def gradient_descent_demo(xs, ys, eta, epoci=45):
    xs, ys = np.asarray(xs, dtype=float), np.asarray(ys, dtype=float)
    w, b = 0.0, 0.0
    urme = []
    for t in range(epoci+1):
        e = w*xs + b - ys
        urme.append((t,w,b,float(np.mean(e**2))))
        gw = 2*np.mean(e*xs)
        gb = 2*np.mean(e)
        w_nou = w - eta*gw
        b_nou = b - eta*gb
        w,b = w_nou,b_nou
    return np.array(urme)

fig, ax = plt.subplots(figsize=(8,4.7))
for eta in [.005,.05,.25]:
    urma = gradient_descent_demo([1,2,3],[2,3,5],eta)
    ax.semilogy(urma[:,0],urma[:,3],label=f"η={eta}")

from sklearn.pipeline import make_pipeline
from sklearn.preprocessing import PolynomialFeatures, StandardScaler
from sklearn.linear_model import Ridge
from sklearn.metrics import mean_squared_error

rng = np.random.default_rng(42)
X_train = np.linspace(-1,1,21).reshape(-1,1)
X_val = np.linspace(-.97,.97,16).reshape(-1,1)
relatie = lambda x: 1.4+.4*x+1.5*x*x

10 · Train, validare și test: o procedură corectă Cod Python ▾
from sklearn.model_selection import train_test_split, GridSearchCV
from sklearn.pipeline import Pipeline
from sklearn.preprocessing import PolynomialFeatures, StandardScaler
from sklearn.linear_model import Ridge
from sklearn.metrics import mean_absolute_error, mean_squared_error, r2_score

rng=np.random.default_rng(17)
X=rng.uniform(-2,2,size=(150,1))
y=1.2+.7*X[:,0]+1.1*X[:,0]**2+rng.normal(0,.3,len(X))
X_tr,X_te,y_tr,y_te=train_test_split(X,y,test_size=.2,random_state=7)
pipe=Pipeline([('poly',PolynomialFeatures(include_bias=False)),
               ('scaler',StandardScaler()),('model',Ridge())])
search=GridSearchCV(pipe,{'poly__degree':[1,2,3,5],
                          'model__alpha':[.01,.1,1,10]},cv=5,scoring='neg_mean_squared_error')
search.fit(X_tr,y_tr)
pred=search.predict(X_te)
print('Parametri aleși exclusiv prin CV pe train:',search.best_params_)
print(f'TEST: MAE={mean_absolute_error(y_te,pred):.3f}, '
      f'RMSE={np.sqrt(mean_squared_error(y_te,pred)):.3f}, '
      f'R²={r2_score(y_te,pred):.3f}')
